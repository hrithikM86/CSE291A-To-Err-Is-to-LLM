def triangle_area(a, h):
    """
    Given length of a side and high return area for a triangle.
    >>> triangle_area(5, 3)
    15.0
    """
    if a < 0 or h < 0:
        return "Invalid input: side and height must be non-negative."
    if a == 0 or h == 0:
        return 0.0
    return a * h
