def encode(message: str) -> str:
    vowels = 'aeiou'
    encoded_message = []
    
    for char in message:
        if char.isalpha():
            # Swap case
            swapped_char = char.swapcase()
            # Check if it's a vowel and shift by one position
            if swapped_char.lower() in vowels:
                if swapped_char.islower():
                    new_char = 'a' if swapped_char == 'e' else 'e' if swapped_char == 'i' else 'i' if swapped_char == 'o' else 'o' if swapped_char == 'u' else 'u'
                else:
                    new_char = 'A' if swapped_char == 'E' else 'E' if swapped_char == 'I' else 'I' if swapped_char == 'O' else 'O' if swapped_char == 'U' else 'U'
                encoded_message.append(new_char)
            else:
                encoded_message.append(swapped_char)
        else:
            encoded_message.append(char)
    
    return ''.join(encoded_message)
