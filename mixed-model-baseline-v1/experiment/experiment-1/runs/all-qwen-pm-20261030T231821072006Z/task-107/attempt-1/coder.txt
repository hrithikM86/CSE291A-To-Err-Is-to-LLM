def is_palindrome(num: int) -> bool:
    return str(num) == str(num)[::-1]

def even_odd_palindrome(n: int) -> tuple:
    if n < 1 or n > 10**3:
        raise ValueError("n must be between 1 and 10^3 inclusive")
    
    even_count = 1  # 0 is an even palindrome
    odd_count = 0
    
    for i in range(1, n + 1):
        if is_palindrome(i):
            if i % 2 == 0:
                even_count += 1
            else:
                odd_count += 1
    
    return (even_count, odd_count)

# Testing the function
print(even_odd_palindrome(3))  # Output: (1, 2)
print(even_odd_palindrome(12)) # Output: (4, 6)
print(even_odd_palindrome(1))  # Output: (1, 1)
print(even_odd_palindrome(2))  # Output: (2, 1)
