def match_parens(lst: list) -> str:
    if len(lst) != 2:
        return 'No'
    
    def is_balanced(s: str) -> bool:
        balance = 0
        for char in s:
            if char == '(':
                balance += 1
            else:
                balance -= 1
            if balance < 0:
                return False
        return balance == 0
    
    str1, str2 = lst[0], lst[1]
    
    if not str1 and is_balanced(str2):
        return 'Yes'
    if not str2 and is_balanced(str1):
        return 'Yes'
    if is_balanced(str1 + str2):
        return 'Yes'
    
    return 'No'
