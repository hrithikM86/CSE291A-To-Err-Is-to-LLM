def tri(n: int) -> list:
    if n < 0:
        raise ValueError("n must be a non-negative integer")

    if n == 0:
        return [3]
    elif n == 1:
        return [3, 2]

    sequence = [3, 2]
    for i in range(2, n + 1):
        if i % 2 == 0:
            sequence.append(1 + i // 2)
        else:
            sequence.append(sequence[-1] + sequence[-2] + 3)

    return sequence[:n + 1]
