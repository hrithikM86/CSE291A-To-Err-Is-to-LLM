```python
def simplify(x: str, n: str) -> bool:
    import re

    # Regular expression to match the fraction format
    fraction_pattern = re.compile(r"^\d+/\d+$")

    # Check if the input strings match the fraction format
    if not (fraction_pattern.match(x) and fraction_pattern.match(n)):
        raise ValueError("Input strings must be in the format <numerator>/<denominator>")

    # Split the fractions into numerator and denominator
    x_num, x_den = map(int, x.split('/'))
    n_num, n_den = map(int, n.split('/'))

    # Simplify the expression x * n
    result_num = x_num * n_num
    result_den = x_den * n_den

    # Check if the result is a whole number
    return result_num % result_den == 0

# Example Implementations
print(simplify("1/5", "5/1"))  # True
print(simplify("1/6", "2/1"))  # False
print(simplify("7/10", "10/2"))  # False
print(simplify("1/1", "2/2"))  # True
```
