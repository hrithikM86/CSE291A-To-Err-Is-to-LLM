def is_prime(n: int) -> bool:
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def words_in_sentence(sentence: str) -> str:
    if not sentence:
        return ""
    
    words = sentence.split()
    prime_length_words = []
    
    for word in words:
        length = len(word)
        if is_prime(length):
            prime_length_words.append(word)
    
    return " ".join(prime_length_words)
