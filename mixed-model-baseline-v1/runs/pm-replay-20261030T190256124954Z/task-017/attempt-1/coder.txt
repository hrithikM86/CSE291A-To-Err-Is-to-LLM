```python
from typing import List

def parse_music(music_string: str) -> List[int]:
    durations = []
    if not music_string:
        return durations
    
    note_map = {'o': 4, 'o|': 2, '.|': 1}
    
    for note in music_string.split():
        if note in note_map:
            durations.append(note_map[note])
    
    return durations
```
