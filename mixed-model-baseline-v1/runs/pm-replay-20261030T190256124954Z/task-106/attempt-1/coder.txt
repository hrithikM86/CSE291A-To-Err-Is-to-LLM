def f(n: int) -> list:
    if n <= 0:
        return []
    
    result = []
    
    for i in range(1, n + 1):
        if i % 2 == 0:
            result.append(factorial(i))
        else:
            result.append(sum_to(i))
    
    return result

def factorial(x: int) -> int:
    if x == 0:
        return 1
    result = 1
    for i in range(1, x + 1):
        result *= i
    return result

def sum_to(x: int) -> int:
    return sum(range(1, x + 1))
