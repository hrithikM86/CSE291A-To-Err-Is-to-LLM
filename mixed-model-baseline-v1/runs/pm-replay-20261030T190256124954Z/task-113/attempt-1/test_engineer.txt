def odd_count(lst):
    result = []
    for i, s in enumerate(lst):
        if not s:
            result.append("")
            continue
        even_count = sum(1 for char in s if int(char) % 2 == 0)
        odd_count = len(s) - even_count
        output_string = f"the number of odd elements {odd_count}n the str{odd_count}ng {odd_count} of the {odd_count}nput."
        result.append(output_string)
    return result
